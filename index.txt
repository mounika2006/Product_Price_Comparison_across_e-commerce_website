    <script>
        // Simple client-side search & sort for the table
        (function(){
            const table = document.getElementById('resultsTable');
            if(!table) return;
            const tbody = table.querySelector('tbody');
            const rows = Array.from(tbody.querySelectorAll('tr'));

            // Search box is optional; if present it will filter
            const searchBox = document.getElementById('searchBox');
            searchBox && searchBox.addEventListener('input', () => {
                const q = searchBox.value.trim().toLowerCase();
                rows.forEach(r => {
                    const txt = r.innerText.toLowerCase();
                    r.style.display = txt.includes(q) ? '' : 'none';
                });
            });

            // Sort
            const headers = Array.from(table.querySelectorAll('thead th[data-key]'));
            headers.forEach(h => {
                h.addEventListener('click', () => {
                    const key = h.getAttribute('data-key');
                    const dir = h._dir = -(h._dir || -1); // toggle
                    const sorted = rows.slice().sort((a,b) => {
                        const aVal = cellValue(a, key);
                        const bVal = cellValue(b, key);
                        if (aVal < bVal) return -1 * dir;
                        if (aVal > bVal) return 1 * dir;
                        return 0;
                    });
                    // re-append
                    sorted.forEach(r => tbody.appendChild(r));
                });
            });

            function cellValue(row, key){
                try{
                    if(key === 'platform'){
                        return row.querySelector('.platform-badge')?.innerText.trim() || '';
                    }
                    if(key === 'title'){
                        return row.querySelector('td:nth-child(2)')?.innerText.trim() || '';
                    }
                    if(key === 'price'){
                        const txt = row.querySelector('td:nth-child(3)')?.innerText.replace(/[^0-9.]/g,'') || '';
                        return parseFloat(txt) || 0;
                    }
                    if(key === 'ship'){
                        // ship date sorting: attempt to parse a YYYY-MM-DD date, else return large string fallback
                        const txt = row.querySelector('td:nth-child(4)')?.innerText.trim() || '';
                        const iso = txt.match(/\\d{4}-\\d{2}-\\d{2}/);
                        if(iso) return iso[0];
                        // if text like "Ships within 2-4 days" transform to a small integer estimate (days)
                        const daysMatch = txt.match(/(\\d+)[^\\d]*-(\\d+)/);
                        if(daysMatch) return parseInt(daysMatch[1],10);
                        const singleDay = txt.match(/(\\d+)\\s*days?/i);
                        if(singleDay) return parseInt(singleDay[1],10);
                        return '9999'; // unknown -> sort last
                    }
                    if(key === 'rating'){
                        const txt = row.querySelector('td:nth-child(5)')?.innerText.replace(/[^0-9.]/g,'') || '';
                        return parseFloat(txt) || 0;
                    }
                    if(key === 'reviews'){
                        const txt = row.querySelector('td:nth-child(6)')?.innerText.replace(/[^0-9]/g,'') || '';
                        return parseInt(txt) || 0;
                    }
                } catch(e){ return '' }
            }
        })();
    </script>
